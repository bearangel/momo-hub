## Review Focus（五类最可能咬人的输入——每行测试钉在所属任务）

1. **未镜像空间的路由降级**：用户在未入组镜像的工作空间打开新建弹窗 → 执行位置区只有「本机」，无灰选项无报错 → Task 8 测

4. **超时后永卡 running**：denied+steer 注入后 turn 必须收尾或改道，账本不得滞留 running（D6 同哲学）→ Task 11 测
5. **退役残留死入口**：DelegationPanel/Wizard/旧 ViewPanel 摘除后，设置页与组卡零死链接、旧 store 订阅全部清理、被退役 IPC 零引用 → Task 5 测

## 接口契约（跨任务类型锁——后续任务按此引用，不得改名）

```ts
// ── 路由（Task 7 定义，9/10 消费）──
type TaskRouting =
  | { mode: 'local' }
  | { mode: 'group'; groupId: string }
  | { mode: 'node'; groupId: string; targetNodeId: string };
// tasks 表迁移：+ routing_json TEXT NULL、+ delegation_id TEXT NULL（本地库，非 hub 协议）

// ── 调度器（Task 8 定义，9 消费；纯函数）──
interface DispatchInput {
  groupId: string;
  workspaceName: string;
  /** 在线成员（presence 过滤后），runningCount=该节点活跃委派数 */
  members: Array<{ nodeId: string; name: string; online: boolean; runningCount: number }>;
  /** 已镜像该工作空间的节点 id 集 */
  mirroredBy: Set<string>;
}
function pickWorker(input: DispatchInput): string | null; // null=无候选

// ── HITL 端口（Task 12/13 定义）──
type PermissionTier = 'readonly' | 'sandbox' | 'confirm';
// DelegationRunPorts 扩展：
applyRunPolicy(sessionId: string, tier: PermissionTier): void;          // Task 12
sessionWaiting(sessionId: string): { waiting: boolean; summary: string } | null; // Task 13
// task.report 进度载荷扩展（向后兼容可选字段）：
progress: { lastSeq: number; excerpt: string; kind?: 'waiting-input' }   // Task 13
// 超时设置（节点互联·委派节，Task 13）：
hitlTimeoutMinutes: number;  // 默认 15
```
